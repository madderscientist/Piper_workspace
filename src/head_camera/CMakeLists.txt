cmake_minimum_required(VERSION 3.16)
project(head_camera)

find_package(ament_cmake REQUIRED)
find_package(k4a 1.4 CONFIG QUIET)
find_package(k4arecord 1.4 CONFIG QUIET)
find_package(OpenCV REQUIRED COMPONENTS core imgcodecs imgproc calib3d aruco)
find_package(yaml-cpp REQUIRED)
find_package(Boost REQUIRED COMPONENTS system)
find_package(Threads REQUIRED)
find_package(ament_index_cpp REQUIRED)

add_library(head_camera_core STATIC src/options.cpp src/storage.cpp src/acquisition.cpp src/calibration.cpp src/web.cpp)
target_compile_features(head_camera_core PUBLIC cxx_std_17)
target_include_directories(head_camera_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(head_camera_core PUBLIC
  yaml-cpp ${OpenCV_LIBS} Boost::system Threads::Threads)
if(k4a_FOUND AND k4arecord_FOUND)
  target_compile_definitions(head_camera_core PUBLIC HEAD_CAMERA_HAS_K4A=1)
  target_link_libraries(head_camera_core PUBLIC k4a::k4a k4a::k4arecord)
else()
  message(STATUS "Azure Kinect SDK not found; building head_camera with V4L2 pure-RGB support only")
  target_compile_definitions(head_camera_core PUBLIC HEAD_CAMERA_HAS_K4A=0)
endif()
ament_target_dependencies(head_camera_core PUBLIC ament_index_cpp)
target_compile_options(head_camera_core PRIVATE -Wall -Wextra -Wpedantic)

add_executable(head_camera src/main.cpp)
target_link_libraries(head_camera PRIVATE head_camera_core)
set_target_properties(head_camera PROPERTIES INSTALL_RPATH_USE_LINK_PATH TRUE)
target_compile_options(head_camera PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS head_camera DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config web DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_head_camera test/test_head_camera.cpp test/test_web.cpp)
  target_link_libraries(test_head_camera head_camera_core)
  # The real recording library must also use the test-only device functions.
  target_link_options(test_head_camera PRIVATE -Wl,--export-dynamic)
endif()

ament_package()
